<script setup lang="ts">
import { computed, onMounted, reactive, ref, watch } from 'vue';

import { getErrorMessage } from '@/utils/errorMessage';

import EmptyState from '@/components/common/EmptyState.vue';
import ErrorState from '@/components/common/ErrorState.vue';
import PageHeader from '@/components/common/PageHeader.vue';
import StatusTag from '@/components/common/StatusTag.vue';
import DifficultyTag from '@/components/question/DifficultyTag.vue';
import QuestionPreviewCard from '@/components/question/QuestionPreviewCard.vue';
import QuestionTypeTag from '@/components/question/QuestionTypeTag.vue';
import { DIFFICULTY_META, QUESTION_TYPE_META } from '@/constants/question';
import { QUESTION_STATUS_META } from '@/constants/status';
import { listVisibleCourses } from '@/services/academicService';
import { addQuestionToPaper, listPapers } from '@/services/paperService';
import {
  listQuestionsPage,
  updateQuestion,
  type QuestionUpsertRequest,
} from '@/services/questionService';
import type { Course } from '@/types/academic';
import type { Paper } from '@/types/paper';
import type {
  Difficulty,
  Question,
  QuestionQuery,
  QuestionSource,
  QuestionStatus,
  QuestionType,
} from '@/types/question';

type FilterState = Required<
  Pick<QuestionQuery, 'keyword' | 'type' | 'difficulty' | 'status' | 'source' | 'courseId'>
>;

type EditQuestionForm = {
  courseId: string;
  title: string;
  type: QuestionType;
  difficulty: Difficulty;
  knowledgePoints: string;
  score: number;
  answer: string;
  analysis: string;
  optionsJson: string;
  scoringPointsJson: string;
};

const SOURCE_META: Record<QuestionSource, string> = {
  MANUAL: '浜哄伐褰曞叆',
  AI: 'AI 鐢熸垚',
  IMPORT: '鎵归噺瀵煎叆',
};

const typeOptions = Object.entries(QUESTION_TYPE_META).map(([value, meta]) => ({
  value: value as QuestionType,
  label: meta.label,
}));
const difficultyOptions = Object.entries(DIFFICULTY_META).map(([value, meta]) => ({
  value: value as Difficulty,
  label: meta.label,
}));
const statusOptions = Object.entries(QUESTION_STATUS_META).map(([value, meta]) => ({
  value: value as QuestionStatus,
  label: meta.label,
}));
const sourceOptions = Object.entries(SOURCE_META).map(([value, label]) => ({
  value: value as QuestionSource,
  label,
}));
const pageSizeOptions = [10, 20, 50];

const query = reactive<FilterState>({
  keyword: '',
  courseId: 'ALL',
  type: 'ALL',
  difficulty: 'ALL',
  status: 'ALL',
  source: 'ALL',
});

const pagination = reactive({
  page: 1,
  pageSize: 10,
  total: 0,
});

const courses = ref<Course[]>([]);
const questions = ref<Question[]>([]);
const selectedQuestion = ref<Question | null>(null);
const loading = ref(false);
const errorMessage = ref<string | null>(null);
const isFilterCollapsed = ref(true);
const feedbackMessage = ref<string | null>(null);
const feedbackTone = ref<'success' | 'danger' | 'info'>('info');

const papers = ref<Paper[]>([]);
const paperLoading = ref(false);
const composingQuestion = ref<Question | null>(null);
const selectedPaperId = ref('');
const editSaving = ref(false);
const composingSaving = ref(false);
const editingQuestion = ref<Question | null>(null);
const editForm = reactive<EditQuestionForm>({
  courseId: '',
  title: '',
  type: 'SINGLE_CHOICE',
  difficulty: 'MEDIUM',
  knowledgePoints: '',
  score: 0,
  answer: '',
  analysis: '',
  optionsJson: '[]',
  scoringPointsJson: '[]',
});

const totalPages = computed(() => Math.max(Math.ceil(pagination.total / pagination.pageSize), 1));
const pageRangeText = computed(() => {
  if (pagination.total === 0) {
    return '0-0';
  }
  const start = (pagination.page - 1) * pagination.pageSize + 1;
  const end = Math.min(pagination.page * pagination.pageSize, pagination.total);

  return `${start}-${end}`;
});
const availablePapers = computed(() => {
  if (!composingQuestion.value) {
    return [];
  }

  return papers.value.filter(
    (paper) =>
      paper.courseId === composingQuestion.value?.courseId &&
      (paper.status === 'DRAFT' || paper.status === 'READY'),
  );
});

const showFeedback = (message: string, tone: 'success' | 'danger' | 'info' = 'info') => {
  feedbackMessage.value = message;
  feedbackTone.value = tone;
};

const clearFeedback = () => {
  feedbackMessage.value = null;
};

const loadQuestions = async () => {
  loading.value = true;
  errorMessage.value = null;

  try {
    const result = await listQuestionsPage({
      ...query,
      page: pagination.page,
      pageSize: pagination.pageSize,
    });
    questions.value = result.items;
    pagination.total = result.total;
    selectedQuestion.value =
      result.items.find((question) => question.id === selectedQuestion.value?.id) ??
      result.items[0] ??
      null;
    if (pagination.page > totalPages.value) {
      pagination.page = totalPages.value;
    }
  } catch (error) {
    questions.value = [];
    selectedQuestion.value = null;
    pagination.total = 0;
    errorMessage.value = getErrorMessage(error, '棰樺簱鍔犺浇澶辫触锛岃绋嶅悗閲嶈瘯');
  } finally {
    loading.value = false;
  }
};

const loadCourses = async () => {
  try {
    courses.value = await listVisibleCourses();
  } catch {
    courses.value = [];
  }
};

const resetFilters = () => {
  query.keyword = '';
  query.courseId = 'ALL';
  query.type = 'ALL';
  query.difficulty = 'ALL';
  query.status = 'ALL';
  query.source = 'ALL';
};

const setPage = (page: number) => {
  pagination.page = Math.min(Math.max(page, 1), totalPages.value);
};

const selectQuestion = (question: Question) => {
  selectedQuestion.value = question;
};

const formatJson = (value: unknown) => JSON.stringify(value ?? [], null, 2);

const openEdit = (question: Question) => {
  clearFeedback();
  selectedQuestion.value = question;
  editingQuestion.value = question;
  editForm.courseId = question.courseId;
  editForm.title = question.title;
  editForm.type = question.type;
  editForm.difficulty = question.difficulty;
  editForm.knowledgePoints = question.knowledgePoints.join('锛?);
  editForm.score = question.score;
  editForm.answer = Array.isArray(question.answer) ? question.answer.join(',') : question.answer;
  editForm.analysis = question.analysis;
  editForm.optionsJson = formatJson(question.options);
  editForm.scoringPointsJson = formatJson(question.scoringPoints);
};

const closeEdit = () => {
  editingQuestion.value = null;
};

const parseList = (value: string) =>
  value
    .split(/[锛?\n]/)
    .map((item) => item.trim())
    .filter(Boolean);

const parseJsonArray = <T,>(value: string, label: string): T[] => {
  const parsed = JSON.parse(value || '[]') as unknown;
  if (!Array.isArray(parsed)) {
    throw new Error(`${label} 蹇呴』鏄?JSON 鏁扮粍`);
  }

  return parsed as T[];
};

const buildUpdateRequest = (): QuestionUpsertRequest => ({
  courseId: editForm.courseId,
  title: editForm.title.trim(),
  type: editForm.type,
  difficulty: editForm.difficulty,
  knowledgePoints: parseList(editForm.knowledgePoints),
  score: Number(editForm.score) || 0,
  answer: editForm.type === 'MULTIPLE_CHOICE' ? parseList(editForm.answer) : editForm.answer.trim(),
  analysis: editForm.analysis.trim(),
  options: parseJsonArray<QuestionUpsertRequest['options'][number]>(editForm.optionsJson, '閫夐」'),
  scoringPoints: parseJsonArray<QuestionUpsertRequest['scoringPoints'][number]>(
    editForm.scoringPointsJson,
    '璇勫垎鐐?,
  ),
});

const saveEdit = async () => {
  if (!editingQuestion.value) {
    return;
  }
  editSaving.value = true;
  try {
    const updated = await updateQuestion(editingQuestion.value.id, buildUpdateRequest());
    selectedQuestion.value = updated;
    editingQuestion.value = null;
    showFeedback('棰樼洰宸蹭繚瀛橈紱濡備负宸查┏鍥?AI 棰橈紝灏嗚嚜鍔ㄩ噸鏂拌繘鍏ュ緟瀹℃牳銆?, 'success');
    await loadQuestions();
  } catch (error) {
    showFeedback(getErrorMessage(error, '棰樼洰淇濆瓨澶辫触锛岃妫€鏌ヨ〃鍗曞唴瀹?), 'danger');
  } finally {
    editSaving.value = false;
  }
};

const loadPaperOptions = async () => {
  paperLoading.value = true;
  try {
    papers.value = await listPapers();
  } catch (error) {
    showFeedback(getErrorMessage(error, '璇曞嵎鍒楄〃鍔犺浇澶辫触'), 'danger');
  } finally {
    paperLoading.value = false;
  }
};

const openCompose = async (question: Question) => {
  clearFeedback();
  selectedQuestion.value = question;
  composingQuestion.value = question;
  selectedPaperId.value = '';
  await loadPaperOptions();
  selectedPaperId.value = availablePapers.value[0]?.id ?? '';
};

const closeCompose = () => {
  composingQuestion.value = null;
  selectedPaperId.value = '';
};

const confirmCompose = async () => {
  if (!composingQuestion.value || !selectedPaperId.value) {
    return;
  }
  composingSaving.value = true;
  try {
    await addQuestionToPaper(selectedPaperId.value, composingQuestion.value);
    showFeedback('棰樼洰宸插姞鍏ヨ瘯鍗凤紝鍙墠寰€鈥滆瘯鍗蜂笌缁勫嵎鈥濈户缁皟鏁撮搴忓拰鍒嗗€笺€?, 'success');
    closeCompose();
  } catch (error) {
    showFeedback(getErrorMessage(error, '鍔犲叆璇曞嵎澶辫触锛岃纭璇剧▼涓庤瘯鍗蜂竴鑷?), 'danger');
  } finally {
    composingSaving.value = false;
  }
};

onMounted(() => {
  void loadCourses();
});

watch(
  query,
  () => {
    if (pagination.page !== 1) {
      pagination.page = 1;
      return;
    }
    void loadQuestions();
  },
  { deep: true, immediate: true },
);

watch(
  () => pagination.pageSize,
  () => {
    if (pagination.page !== 1) {
      pagination.page = 1;
      return;
    }
    void loadQuestions();
  },
);

watch(
  () => pagination.page,
  () => {
    void loadQuestions();
  },
);
</script>
